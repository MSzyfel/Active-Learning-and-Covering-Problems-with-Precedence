\begin{algorithm}[h]
\DontPrintSemicolon
\caption{The algorithm for $\ProblemPCMSSC$ based on an $\br{\alpha,\beta}$-approximation for $\ProblemBPCSC$.}
\label{alg:fPCMSSC}
\SetKwProg{Proc}{procedure}{}{}
\Proc{$\ProcedureFPCMSSC(\cU,\cS,\preceq,f)$}{
    $\cQ\gets\emptyset$\;
    \For{$1\leq l\leq\max\brc{1,\cl{\log_a\max\brc{1,\spr{\cS}}}}$}{
        $\cU'\gets\cU$\;
        $\pi_l\gets\emptyset$\;
        \For{$1\leq i\leq l$}{
            $\cC_l^i\gets$ Run the $\br{\alpha,\beta}$-approximation for $\ProblemBPCSC$ on $(\cU',\cS,\preceq,\budget=a^{i+1})$.\;
            $\cU'\gets \cU'\setminus\cov(\cC_l^i)$\;
            $\pi_l\gets\pi_l\circ\operatorname{ord}(\cC_l^i\setminus \pi_l)$\;
        }
        $T_l\gets\min\brc{m,\cl{\alpha\cdot\beta\cdot a^l}}$\;
        \While{$\spr{\pi_l}<T_l$}{
            $S_l\gets$ the next set in the fixed topological ordering of $\cS$ that is not in $\pi_l$.\;
            $\pi_l\gets\pi_l\circ\angl{S_l}$\;
        }
        $\cR_l\gets\cU\setminus\cov(\pi_l)$\;
        $\cC_l\gets$ Run the $\br{\alpha,\beta}$-approximation for $\ProblemBPCSC$ on $(\cR_l,\cS,\preceq,\budget=a^l)$.\;
        $\pi_l\gets\pi_l\circ\operatorname{ord}(\cC_l\setminus \pi_l)$\;
        \If{$\spr{\cov\br{\pi_l}}\geq f\cdot\spr{\cU}/\beta$}{
            $\cQ\gets\cQ\cup\brc{\pi_l}$\;
        }
    }
    \Return $\argmin_{\pi\in\cQ}\brc{\coverageTime\br{\pi}}$\;
}
\end{algorithm}
