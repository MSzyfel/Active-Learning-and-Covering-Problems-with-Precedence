\begin{algorithm}[h]
\DontPrintSemicolon
\LinesNumbered
\caption{An $\alpha_{m,n}$-approximation algorithm for $\ProblemPCSC$}
\label{alg:PCSC}
\SetKwProg{Proc}{procedure}{}{}
\Proc{$\ProcedurePCSC(\cU, \cS, \preceq, k)$}{
  $\cR\gets\brc{r_S\colon S\in\cS}$\;
  $\cB\gets\brc{b_u\colon u\in\cU}$\;
  $\cA\gets\emptyset$\;

  \ForEach{$S\in\cS$}{
    $\cS_S\gets\closure{S}$\;
    $A_S\gets\brc{r_T\colon T\in \cS_S}\cup\brc{b_u\colon u\in\cov\br{\cS_S}}$\;
    $\cA\gets\cA\cup\brc{A_S}$\;
  }

  $\cA'\gets$ Run the $\alpha_{m,n}$-approximation algorithm for $\ProblemPRBSC$ on $(\cR,\cB,\cA,k)$\;
  $\cC\gets\bigcup_{A_S\in\cA'}\cS_S$\;
  \Return $\cC$\;
}
\end{algorithm}
