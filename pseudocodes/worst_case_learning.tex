\begin{algorithm}[h]
\DontPrintSemicolon
\caption{A generic testing strategy for $\ProblemPCACDT$ and $\ProblemPCWCDT$.}
\label{alg:worstDecisionTree}
\SetKwProg{Proc}{procedure}{}{}
\Proc{\ProcDecisionTree$(\cH,\cT,\preceq,\blackBox)$}{
    \If{$\spr{\cH}=1$}{
        \Return the only hypothesis in $\cH$.\;
    }
    % Construct $\xi$, $\cU_{\xi}$, and $\preceq_{\xi}$ from the restricted tests $t(\cH)$ for the current hypothesis set $\cH$.\;
    $\cC\gets$ Call the $\br{\alpha,\beta}$-approximation algorithm $\blackBox$ on $(\cU_{\xi},\xi(\cT),\preceq_{\xi},f=1/4)$.\;
    Order $\cC$ topologically if it is a family
    $\cH'\gets \cH$.\;
    \ForEach{test $t\in \angl{\xi(t_1),\ldots,\xi(t_{\spr{\cC}})}$}{
        Perform $t$ and observe its reply $H$, and let $\cH'\gets\cH'\cap H$\;
        \If{$\spr{\cH'}\leq\frac{5\cdot\beta-1}{5\cdot\beta}\cdot\spr{\cH}$}{
            \Return \ProcDecisionTree$(\cH',\cT,\preceq,\blackBox)$.\;
        }
    }
}
\end{algorithm}
