\section{Set covering with precedence constraints} \label{sec:SC}

% Similarly as in previous section we assume our input instance to be unweighted. Note that the analysis of the following procedures works also in the weighted setting, where every element $u\in\cU$ has a weight $w(u)$ and the goal is to maximize the total weight of covered elements or cover at least a given total weight of elements with minimal size of the cover. Since this generalization is straightforward we omit it for clarity.

Our starting point is the \textup{\textsc{Partial Red-Blue Set Cover}} ($\ProblemPRBSC$) problem.
We first improve an existing algorithm for $\ProblemPRBSC$ and then use it for $\ProblemPCSC$. We then introduce a budgeted variant of precedence-constrained set covering and obtain an approximation for it using $\ProblemPCSC$, which leads to an approximation for $\ProblemPCMSSC$.

\begin{problem}[\textup{\textsc{Partial Red-Blue Set Cover}} ($\ProblemPRBSC$)] \label{problem:PRBSC}
Given disjoint sets $\cR$ of $k$ red elements and $\cB$ of $n$ blue elements, a family $\cS$ of subsets of $\cR\cup\cB$, and an integer target $t$, choose a cover $\cC\subseteq\cS$ such that $\spr{\cov\br{\cC,\cB}}\geq t$ which minimizes the \emph{red cost} $\spr{\cov\br{\cC,\cR}}$.
\end{problem}

\subsection{A bicriteria algorithm for Partial Red-Blue Set Cover}
\label{subsection:PRBSC-bicriteria}
Let $m$ be the number of input sets, $n=\spr{\cB}$ the number of blue elements, and $k=\spr{\cR}$ the number of red elements. Chlamt{\'{a}}{\v{c}}, Makarychev and Vakilian gave a $\bigo\br{m^{1/3}\cdot \log^{4/3}k\cdot \log n}$-approximation for $\ProblemPRBSC$~\cite[Section~3 and Appendix~A]{RedBlueSetCover}. We show that allowing a constant-factor loss in the blue coverage, which is sufficient for our needs, leads to an $\bigo\br{m^{1/3}}$ bound on the red cost.
%  Our deterministic algorithm covers at least one eighth of the target at cost $\bigo\br{m^{1/3}}$ times the optimum for target $t$.
Our algorithm builds on an existing LP-rounding \cite{RedBlueSetCover}, however, we use a stronger LP relaxation enhanced with a non-trivial derandomization step. In the original algorithm, the sets are partitioned according to the number of red elements they contain and a step finding a collection with a favorable ratio of red cost to blue coverage is repeated until the target is met. We inherit this bucketing idea, however, we round all these size groups together. In order to bound the cost of our solution we distinguish \emph{heavy} and \emph{light} covering sets according to the LP solution. Additional constraints for pairs of red elements allow us to bound the expected red cost of heavy sets across all sizes, while light sets have sufficiently small selection probabilities that we can bound their total cost by counting them with multiplicity. In order to show that our algorithm covers a significant fraction of the blue elements, we use random marks to identify some of the covered blue elements. Their randomized choice allows us to lower bound the expected number of distinct marked elements and, at the same time, upper bound this value's second moment. This comes in handy during the derandomization step where we design a carefully defined random variable we call \emph{potential}, whose non-negative value is maintained throughout the rounding using the method of conditional expectations, yielding a quality deterministic solution in polynomial time.

\begin{theorem}\label{thm:partial-red-blue-bicriteria}
For an instance $\cI=\br{\cR,\cB,\cS,t}$ of $\ProblemPRBSC$, there is a deterministic polynomial-time algorithm that outputs a cover $\cC\subseteq\cS$ satisfying $\spr{\cov\br{\cC,\cB}}\geq t/8$ and $\spr{\cov\br{\cC,\cR}}\leq920\cdot m^{1/3}\cdot\OPT\br{\cI}$.
\end{theorem}
\begin{proof}
Write $\cR_S=S\cap\cR$ and $\cB_S=S\cap\cB$ for each $S\in\cS$.
% First select all sets $S$ with $\cR_S=\emptyset$, that is, all sets containing no red elements. If they cover at least $t/8$ blue elements, return this family. In particular, this handles every instance whose optimum red cost is zero.
In the first step we guess an integer $K\in[\spr{\cR}]$ for the optimum red cost $\OPT\br{\cI}$ (this can be done in $O\br{\log\br{\spr{\cR}}}$ tries). Assume that we have a correct guess. In our LP there will be the following variables:
\begin{enumerate}
    \item For each red element $r\in\cR$, the variable $y_r\in[0,1]$ indicating whether $r$ is covered.
    \item For each set $S\in\cS$, the variable $x_S\in[0,1]$ indicating whether $S$ is selected.
    \item For each blue element $u\in\cB$, the variable $z_u\in[0,1]$ indicating whether $u$ is one of the $t$ \emph{chosen} covered blue elements. An optimal integral solution may cover additional blue elements.
    \item For each $S\in\cS$ and $u\in\cB_S$, the variable $a_{u,S}\in[0,1]$ indicating the part of $z_u$ assigned to the set $S$. In an integral solution each chosen $u$  is assigned to one selected set containing it.
    \item For each ordered pair $r,q\in\cR$, the variable $y_{r,q}\in[0,1]$ whose intended integral value is $y_r\cdot y_q$. Note that we will not impose this nonlinear equality, but use linear inequalities that it satisfies.
\end{enumerate}

For the current guess $K$, the LP is as follows:
\begin{align}
 \sum_{r\in\cR}y_r&\leq K, \quad\quad \sum_{u\in\cB}z_u=t, && \label{eq:rbsc-lp-budget}\\
 \sum_{S\in\cS\colon u\in \cB_S}a_{u,S}&=z_u &&(u\in\cB), \label{eq:rbsc-lp-assign}\\
 a_{u,S}&\leq x_S &&(S\in\cS,\ u\in \cB_S), \label{eq:rbsc-lp-marginal}\\
 \sum_{u\in \cB_S}a_{u,S}&\leq t\cdot x_S &&(S\in\cS), \label{eq:rbsc-lp-set}\\
 x_S&\leq y_r &&(S\in\cS,\ r\in \cR_S), \label{eq:rbsc-lp-red}\\
 \sum_{S\in\cS\colon r\in \cR_S,\,u\in \cB_S}a_{u,S}&\leq y_r &&(r\in\cR,\ u\in\cB), \label{eq:rbsc-lp-blue-load}\\
 \sum_{S\in\cS\colon r\in \cR_S}\sum_{u\in \cB_S}a_{u,S}&\leq t\cdot y_r &&(r\in\cR), \label{eq:rbsc-lp-total-load}\\
 x_S&\leq y_{r,q} &&(S\in\cS,\ r,q\in \cR_S), \label{eq:rbsc-lp-pair}\\
 \sum_{q\in\cR}y_{r,q}&\leq K\cdot y_r &&(r\in\cR). \label{eq:rbsc-lp-pair-budget}
\end{align}
% The assignment constraints limit the load on each set and red element. In particular, they prevent the same blue element from contributing too much within a group and bound the total number of marks expected in an active group. The pair constraints give a budget for jointly covered red elements. This will let us charge heavy occurrences across all scales.
%  The following claim verifies the relaxation.

\begin{claim}\label{claim:rbsc-lp-validity}
For the correct guess $K=\OPT\br{\cI}$, the above system is a polynomial-size relaxation of $\ProblemPRBSC$ with red-cost bound $\OPT\br{\cI}$.
\end{claim}
\begin{proof}
Let $\cS^*\subseteq\cS$ be an optimal solution, with $\spr{\cov\br{\cS^*,\cB}}\geq t$ and $\spr{\cov\br{\cS^*,\cR}}=\OPT\br{\cI}=K$. Choose $\cB^*\subseteq\cov\br{\cS^*,\cB}$ with $\spr{\cB^*}=t$ and, for each $u\in\cB^*$, choose a set $S_u\in\cS^*$ containing $u$. Define
\[
 x_S=\boldsymbol{1}_{S\in\cS^*},\qquad
 y_r=\boldsymbol{1}_{r\in\cov\br{\cS^*}},\qquad
 z_u=\boldsymbol{1}_{u\in\cB^*},\qquad
 y_{r,q}=y_r\cdot y_q.
\]
For the assignment variables, set $a_{u,S_u}=1$ for each $u\in\cB^*$ and set all other $a_{u,S}$ to zero.

The budget and assignment constraints hold by construction. Each assigned blue element belongs to a selected set, and at most $t$ blue elements are assigned to any set, proving~\eqref{eq:rbsc-lp-marginal} and~\eqref{eq:rbsc-lp-set}. Each selected set does not contain red elements that are not covered, proving~\eqref{eq:rbsc-lp-red} and~\eqref{eq:rbsc-lp-pair}. If $y_r=0$, no selected set contains $r$, so both assignment sums in~\eqref{eq:rbsc-lp-blue-load} and~\eqref{eq:rbsc-lp-total-load} involving $r$ vanish. If $y_r=1$, a fixed blue element is assigned at most once, and there are $t$ assignments in total. This proves~\eqref{eq:rbsc-lp-blue-load} and~\eqref{eq:rbsc-lp-total-load}. Moreover,
\[
 \sum_{q\in\cR}y_{r,q}=y_r\cdot\sum_{q\in\cR}y_q=\OPT\br{\cI}\cdot y_r=K\cdot y_r,
\]
which proves~\eqref{eq:rbsc-lp-pair-budget}. The numbers of variables and constraints are polynomial in the explicit input size. In particular, the guess $K=\OPT\br{\cI}$ gives a feasible relaxation.
\end{proof}

For the smallest feasible guess $K\leq \OPT\br{\cI}$ fix a feasible solution and put
\[
 \gamma=m^{1/3},\qquad \cR_+=\brc{r\in\cR\colon y_r\geq1/\gamma},\qquad \cR'_S=\cR_S\setminus \cR_+, \textup{ for }S\in\cS.
\]
We call $\cR'_S$ the \emph{residual red part} of $S$. By~\eqref{eq:rbsc-lp-budget}, $\spr{\cR_+}/\gamma\leq\sum_{r\in \cR_+}y_r\leq K$, so $\spr{\cR_+}\leq K\cdot \gamma$. Initially select the family $\cS_0=\brc{S\in\cS\colon \cR'_S=\emptyset}$. Its red coverage satisfies $\cov\br{\cS_0,\cR}\subseteq\cR_+$, and hence $\spr{\cov\br{\cS_0,\cR}}\leq K\cdot \gamma$. If $\spr{\cov\br{\cS_0,\cB}}\geq t/8$, retain $\cS_0$ as the candidate. Otherwise, define $\cB'=\cB\setminus\cov\br{\cS_0}$. Then $\spr{\cB\setminus\cB'}=\spr{\cov\br{\cS_0,\cB}}<t/8$. Since $z_u\leq1$, we obtain
\begin{equation}\label{eq:rbsc-residual-mass}
 \sum_{u\in\cB'}z_u
 =t-\sum_{u\in\cB\setminus\cB'}z_u
 \geq t-\spr{\cB\setminus\cB'}>\frac{7}{8}\cdot t.
\end{equation}

For every power of two $d=2^j\leq\spr{\cR}$ with $j\in\mathbb{Z}_{\geq0}$, which we call a \emph{scale}, define $\cS_d$ and, for each $r\in\cR\setminus \cR_+$ with $y_r>0$, the \emph{group} $\cS_{r,d}$ by
\[
 \cS_d=\brc{S\in\cS\colon d\leq\spr{\cR'_S}<2\cdot d},
 \qquad
 \cS_{r,d}=\brc{S\in\cS_d\colon r\in \cR'_S}.
\]
Each set with a nonempty residual red part belongs to exactly one $\cS_d$. We call $r$ the \emph{pivot} of $\cS_{r,d}$. A set $S\in\cS_d$ belongs to the group $\cS_{r,d}$ for each residual red element $r\in\cR'_S$ with $y_r>0$. We call its membership in each such group an \emph{occurrence} of $S$. A set at scale $d$ has at least $d$ residual red elements and hence, unless its LP value is zero, at least $d$ occurrences.
% This multiplicity will compensate for the factor $1/d$ in the activation probability.

For each $r$ and $d$, let $A_{r,d}$ be the event that group $\cS_{r,d}$ is \emph{activated}, with $\Pr[A_{r,d}]=y_r/d$. For each $S\in\cS_{r,d}$, let $T_{r,d,S}$ be the event that its \emph{trial} succeeds, with $\Pr[T_{r,d,S}]=x_S/y_r$. The activation and trial events are mutually independent, including across groups with the same members. An occurrence of $S$ is \emph{selected} if and only if both $A_{r,d}$ and $T_{r,d,S}$ occur, so selection events in the same group share the activation event. Note that both probabilities belong to $[0,1]$ by~\eqref{eq:rbsc-lp-red} and $d\geq1$. Define the random families $\cF_{r,d}$ and $\cC$, sets $\cR_{r,d}$ and $\cR_0$, and the variable $R$ by
\[
 \begin{gathered}
 \cF_{r,d}=\brc{S\in\cS_{r,d}\colon A_{r,d}\text{ and }T_{r,d,S}\text{ occur}},
 \qquad \cC=\cS_0\cup\bigcup_{r,d}\cF_{r,d},\\
 \cR_{r,d}=\bigcup_{S\in\cF_{r,d}}\cR'_S,
 \qquad \cR_0=\bigcup_{r,d}\cR_{r,d},
 \qquad R=\spr{\cR_+}+\spr{\cR_0}.
 \end{gathered}
\]
The output satisfies $\cov\br{\cC,\cR}\subseteq\cR_+\cup\cR_0$, and therefore $\spr{\cov\br{\cC,\cR}}\leq R$.

\begin{claim}\label{claim:rbsc-red-cost}
The random construction satisfies $\mathbb{E}[R]\leq7\cdot K\cdot \gamma$.
\end{claim}
\begin{proof}
Put $\theta=K/\gamma$. We split occurrences according to their trial probabilities: heavy occurrences can be charged to the joint red variables, while light occurrences are cheap enough to count individually. For each group $\cS_{r,d}$, partition its occurrences into the \emph{heavy} and \emph{light} families
\[
 \cS^h_{r,d}=\brc{S\in\cS_{r,d}\colon x_S/y_r\geq\theta/d},
 \qquad \cS^\ell_{r,d}=\cS_{r,d}\setminus\cS^h_{r,d}.
\]
The same set may be heavy in one group and light in another. Define the random sets of residual red elements reached by selected heavy and light occurrences by
\[
 \cR_h=\bigcup_{r,d}\bigcup_{S\in\cF_{r,d}\cap\cS^h_{r,d}}\cR'_S,
 \qquad
 \cR_\ell=\bigcup_{r,d}\bigcup_{S\in\cF_{r,d}\cap\cS^\ell_{r,d}}\cR'_S.
\]
Define $R_h=\spr{\cR_h}$ and $R_\ell=\spr{\cR_\ell}$. Since every selected occurrence is either heavy or light, $R\leq\spr{\cR_+}+R_h+R_\ell$. We bound the expectations of these two variables separately.

Fix $r,q\in\cR\setminus\cR_+$ with $y_r>0$. Define $\cD_{r,q}=\brc{d\colon \exists S\in\cS^h_{r,d}\colon q\in\cR'_S}$ and the event
\[
 C_{r,q}=\bigcup_{d\in\cD_{r,q}}\ \bigcup_{S\in\cS^h_{r,d}\colon q\in\cR'_S}\br{A_{r,d}\cap T_{r,d,S}}.
\]
Thus, $C_{r,q}$ occurs when a selected heavy occurrence in a group $\cS_{r,d}$ contains $q$ in its residual red part. For all other ordered pairs $r,q\in\cR$, let $C_{r,q}$ be the empty event. If $\cD_{r,q}$ is empty, the event $C_{r,q}$ is impossible. Otherwise, for each $d\in\cD_{r,q}$, a corresponding heavy occurrence satisfies
\[
 y_{r,q}\geq x_S\geq\frac{\theta\cdot y_r}{d},
\]
where the first inequality follows from~\eqref{eq:rbsc-lp-pair}. In particular, $y_{r,q}>0$ and $d\geq\theta\cdot y_r/y_{r,q}$. Let $d_0$ be the smallest scale in $\cD_{r,q}$. All scales are powers of two, so every scale in $\cD_{r,q}$ is of the form $2^j\cdot d_0$ for some integer $j\geq0$. A selected occurrence requires its group to be active, so $C_{r,q}\subseteq\bigcup_{d\in\cD_{r,q}}A_{r,d}$. Applying the union bound and using the geometric series $\sum_{j\geq0}2^{-j}=2$ gives
\[
 \Pr[C_{r,q}]\leq\sum_{d\in\cD_{r,q}}\frac{y_r}{d}
 \leq\sum_{j\geq0}\frac{y_r}{2^j\cdot d_0}=\frac{2\cdot y_r}{d_0}
 \leq\frac{2\cdot y_{r,q}}{\theta}.
\]
The same bound holds when $\cD_{r,q}$ is empty, since $y_{r,q}\geq0$. If $q\in\cR_h$, then some selected heavy occurrence containing $q$ belongs to a group $\cS_{r,d}$, so $C_{r,q}$ occurs for its pivot $r$. Thus the union bound gives $\Pr[q\in\cR_h]\leq\sum_{r\in\cR}\Pr[C_{r,q}]$ for each $q\in\cR$. By linearity of expectation,
\[
 \mathbb{E}[R_h]
 =\sum_{q\in\cR}\Pr[q\in\cR_h]
 \leq\sum_{r\in\cR}\sum_{q\in\cR}\Pr[C_{r,q}]
 \leq\frac{2}{\theta}\cdot\sum_{r\in\cR}\sum_{q\in\cR}y_{r,q}
 \leq\frac{2\cdot K}{\theta}\cdot\sum_{r\in\cR}y_r
 \leq\frac{2\cdot K^2}{\theta}.
\]

Now, for each $S\in\cS^\ell_{r,d}$, the independence of the activation and trial events gives
\[
 \Pr[A_{r,d}\cap T_{r,d,S}]
 =\Pr[A_{r,d}]\cdot\Pr[T_{r,d,S}]
 =\frac{y_r}{d}\cdot\frac{x_S}{y_r}
 =\frac{x_S}{d}<\frac{\theta\cdot y_r}{d^2}.
\]
By linearity of expectation, the above probability bound and $y_r<1/\gamma$,
\[
 \mathbb{E}[R_\ell]
 \leq\sum_{r,d}\sum_{S\in\cS^\ell_{r,d}}\spr{\cR'_S}\cdot\Pr[A_{r,d}\cap T_{r,d,S}]
 \leq\frac{\theta}{\gamma}\cdot\sum_{r,d}\sum_{S\in\cS^\ell_{r,d}}\frac{\spr{\cR'_S}}{d^2}
 \leq\frac{\theta}{\gamma}\cdot\sum_d\sum_{S\in\cS_d}\frac{\spr{\cR'_S}^2}{d^2}
 \leq\frac{4\cdot\theta\cdot m}{\gamma}.
\]
Here, the third inequality counts, for each $S\in\cS_d$, at most one occurrence for each of its $\spr{\cR'_S}$ residual red elements. The last inequality uses $\spr{\cR'_S}/d<2$ and the fact that there are at most $m$ sets. Adding $\spr{\cR_+}\leq K\cdot \gamma$ and substituting $\theta=K/\gamma$ and $m=\gamma^3$ gives
\begin{equation}\label{eq:rbsc-cost-expectation}
 \mathbb{E}[R]
 \leq K\cdot \gamma+\frac{2\cdot K^2}{\theta}+\frac{4\cdot\theta\cdot m}{\gamma}
 =3\cdot K\cdot \gamma+\frac{4\cdot K\cdot m}{\gamma^2}
 =7\cdot K\cdot \gamma.
\end{equation}
\end{proof}

To control blue coverage, we use auxiliary \emph{marks} to identify some of the blue elements covered by selected occurrences. For each occurrence of $S$ with $x_S>0$ in group $\cS_{r,d}$, independently mark each $u\in\cB_S\cap\cB'$ using a coin $Z_{r,d,S,u}$ with $\Pr[Z_{r,d,S,u}=1]=a_{u,S}/x_S$. These marking choices are independent of all activations, trials, and other marking choices. The probabilities lie in $[0,1]$ by~\eqref{eq:rbsc-lp-marginal}. For $x_S=0$, take the coins to be always zero. The marking probability is chosen so that, combined with the trial probability, an occurrence contributes in proportion to its assignment mass $a_{u,S}$. Define $\cM_{r,d,S}=\brc{u\in\cB_S\cap\cB'\colon T_{r,d,S}\text{ occurs and }Z_{r,d,S,u}=1}$. Let $\cM_{r,d}=\bigcup_{S\in\cF_{r,d}}\cM_{r,d,S}$ be the set of elements marked by selected occurrences in group $\cS_{r,d}$ and $\cM=\bigcup_{r,d}\cM_{r,d}$ with $M=\spr{\cM}$.
% In an active group, a successful trial with no marks still selects the occurrence and its red elements.

\begin{claim}\label{claim:rbsc-blue-coverage}
The expected number of distinct marked blue elements satisfies $\mathbb{E}[M]>2\cdot t/5$.
\end{claim}
\begin{proof}
Fix $u\in\cB'$. For an occurrence of $S$ in group $\cS_{r,d}$ with $u\in \cB_S$ and $x_S>0$, the probability that its trial succeeds and its marking coin for $u$ succeeds is
\[
 \Pr[u\in \cM_{r,d,S}]=\frac{x_S}{y_r}\cdot\frac{a_{u,S}}{x_S}=\frac{a_{u,S}}{y_r}.
\]
When $x_S=0$, $\Pr[u\in \cM_{r,d,S}]=a_{u,S}=0$. Let the sum of these probabilities within $\cS_{r,d}$ be
\[
 \lambda_{r,d,u}
 =\sum_{S\in\cS_{r,d}\colon u\in\cB_S}\Pr[u\in\cM_{r,d,S}]
 =\frac1{y_r}\cdot\sum_{S\in\cS_{r,d}\colon u\in \cB_S}a_{u,S}
 \leq1
\]
by~\eqref{eq:rbsc-lp-blue-load}. The choices for the set occurrences are independent. We use the inequality $1-x\leq e^{-x}$ and the concavity of $1-e^{-a\cdot x}$, which gives $1-e^{-a\cdot x}\geq(1-e^{-a})\cdot x$ for $a\geq0$ and $x\in[0,1]$. The group's conditional probability of marking $u$ is therefore at least
\[
 \Pr[u\in\cM_{r,d}\mid A_{r,d}]
 =1-\prod_{S\in\cS_{r,d}\colon u\in \cB_S}
       \br{1-\frac{a_{u,S}}{y_r}}
 \geq1-e^{-\lambda_{r,d,u}}
 \geq(1-e^{-1})\cdot\lambda_{r,d,u}.
\]
Since $\cM_{r,d}=\emptyset$ when the group is inactive, the preceding bound and the definition of $\lambda_{r,d,u}$ give
\[
 \Pr[u\in\cM_{r,d}]=\Pr[A_{r,d}]\cdot\Pr[u\in\cM_{r,d}\mid A_{r,d}]
 \geq\frac{y_r}{d}\cdot(1-e^{-1})\cdot\lambda_{r,d,u}
 =\frac{1-e^{-1}}{d}\cdot\sum_{S\in\cS_{r,d}\colon u\in \cB_S}a_{u,S}.
\]
Let $M_u=\sum_{r,d}\boldsymbol{1}_{u\in\cM_{r,d}}$ be the number of groups that mark $u$. By linearity of expectation and the group bound above,
\begin{align*}
 \mathbb{E}[M_u]&=\sum_{r,d}\Pr[u\in\cM_{r,d}]
 \geq(1-e^{-1})\cdot\sum_{r,d}\frac1d\cdot
       \sum_{S\in\cS_{r,d}\colon u\in \cB_S}a_{u,S}\\
 &=(1-e^{-1})\cdot\sum_d\sum_{S\in\cS_d\colon u\in \cB_S}
       \frac{a_{u,S}}d\cdot\sum_{r\in\cR'_S\colon y_r>0}1\\
 &=(1-e^{-1})\cdot\sum_d\sum_{S\in\cS_d\colon u\in \cB_S}
       \frac{\spr{\cR'_S}}d\cdot a_{u,S}
 \geq(1-e^{-1})\cdot\sum_{S\in\cS\colon u\in \cB_S}a_{u,S}
 =(1-e^{-1})\cdot z_u.
\end{align*}
For the second equality, fix a scale $d$ and a set $S\in\cS_d$ containing $u$. The term $a_{u,S}$ appears in $\sum_r\sum_{S'\in\cS_{r,d}\colon u\in\cB_{S'}}a_{u,S'}$ once for each $r\in\cR'_S$ with $y_r>0$. If $y_r=0$ for some $r\in\cR'_S$, then $a_{u,S}\leq x_S\leq y_r=0$ by~\eqref{eq:rbsc-lp-marginal} and~\eqref{eq:rbsc-lp-red}, so these pivots can also be counted. For the last inequality, every set $S$ containing $u\in\cB'$ has nonempty $\cR'_S$. Otherwise, $S\in\cS_0$ and would cover $u$. Such a set belongs to exactly one scale $d$, where $\spr{\cR'_S}/d\geq1$. Since $a_{u,S}\geq0$, replacing this factor by $1$ gives the lower bound.

The rounding choices for distinct group indices are independent. Applying $1-x\leq e^{-x}$ across groups and then the above concavity bound for $z_u\in[0,1]$ yields
\[
 \Pr[u\in\cM]
 =1-\prod_{r,d}\br{1-\Pr[u\in\cM_{r,d}]}
 \geq1-e^{-\mathbb{E}[M_u]}
 \geq1-e^{-(1-e^{-1})\cdot z_u}
 \geq c\cdot z_u,
 \qquad c=1-e^{-(1-e^{-1})}.
\]
Since $c>23/50$, summing over $\cB'$ and using~\eqref{eq:rbsc-residual-mass} gives
\[
 \mathbb{E}[M]=\sum_{u\in\cB'}\Pr[u\in\cM]\geq c\cdot\sum_{u\in\cB'}z_u
 >\frac{23}{50}\cdot\frac{7\cdot t}{8}
 =\frac{161\cdot t}{400}>\frac{2\cdot t}{5}.
\]
\end{proof}

For each group $\cS_{r,d}$, define
\[
 M_{r,d}=\sum_{S\in\cS_{r,d}}\spr{\cM_{r,d,S}},
 \qquad W_{r,d}=\boldsymbol{1}_{A_{r,d}}\cdot M_{r,d},
 \qquad W=\sum_{r,d}W_{r,d}.
\]
Thus $W_{r,d}$ counts the marks in group $\cS_{r,d}$ with multiplicity over its selected occurrences. A blue element marked by two occurrences is counted twice in $W$, even if the occurrences are in the same group. In particular, $M\leq W$.

\begin{claim}\label{claim:rbsc-second-moment}
We have $\mathbb{E}[W^2]\leq10\cdot t^2$.
\end{claim}
\begin{proof}
Fix a group $\cS_{r,d}$. For each occurrence $S\in\cS_{r,d}$, let $M_S=\spr{\cM_{r,d,S}}$, so $M_{r,d}=\sum_{S\in\cS_{r,d}}M_S$. These variables are defined before activation and are independent across occurrences. By linearity of expectation and the marking probabilities calculated above,
\[
 \mathbb{E}[M_S]=\sum_{u\in\cB_S\cap\cB'}\Pr[u\in\cM_{r,d,S}]
 =\frac1{y_r}\cdot\sum_{u\in \cB_S\cap\cB'}a_{u,S}.
\]
For the second moment, put $\mu_S=\sum_{u\in\cB_S\cap\cB'}\Pr[Z_{r,d,S,u}=1]$. Let $N_S=\sum_{u\in\cB_S\cap\cB'}Z_{r,d,S,u}$. By~\eqref{eq:rbsc-lp-set}, $\mu_S\leq t$. The marking coins are independent Bernoulli variables, so
\[
 \mathbb{E}[N_S]=\mu_S,\qquad
 \operatorname{Var}(N_S)=\sum_{u\in\cB_S\cap\cB'}\Pr[Z_{r,d,S,u}=1]\cdot\br{1-\Pr[Z_{r,d,S,u}=1]}.
\]
Moreover, $M_S=\boldsymbol{1}_{T_{r,d,S}}\cdot N_S$ and the trial is independent of $N_S$, so $\mathbb{E}[M_S]=(x_S/y_r)\cdot \mu_S$. Using $\mathbb{E}[N_S^2]=\mathbb{E}[N_S]^2+\operatorname{Var}(N_S)$ gives
\[
 \mathbb{E}[M_S^2]=\frac{x_S}{y_r}\cdot\br{\mu_S^2+\sum_{u\in\cB_S\cap\cB'}\Pr[Z_{r,d,S,u}=1]\cdot\br{1-\Pr[Z_{r,d,S,u}=1]}}
 \leq\frac{x_S}{y_r}\cdot(\mu_S^2+\mu_S)
 \leq(t+1)\cdot\mathbb{E}[M_S].
\]
The first-moment calculation and $r\in\cR_S$ for every $S\in\cS_{r,d}$ give
\[
 \mathbb{E}[M_{r,d}]=\frac1{y_r}\cdot\sum_{S\in\cS_{r,d}}\sum_{u\in\cB_S\cap\cB'}a_{u,S}
 \leq\frac1{y_r}\cdot\sum_{S\in\cS\colon r\in\cR_S}\sum_{u\in\cB_S}a_{u,S}\leq t,
\]
where the last inequality follows from~\eqref{eq:rbsc-lp-total-load}. Expanding $M_{r,d}^2$ and using independence across distinct occurrences gives
\begin{align*}
 \mathbb{E}[M_{r,d}^2]
 &=\sum_{S\in\cS_{r,d}}\mathbb{E}[M_S^2]+\sum_{S,S'\in\cS_{r,d}\colon S\ne S'}\mathbb{E}[M_S]\cdot\mathbb{E}[M_{S'}]\\
 &\leq(t+1)\cdot\sum_{S\in\cS_{r,d}}\mathbb{E}[M_S]+\br{\sum_{S\in\cS_{r,d}}\mathbb{E}[M_S]}^2
 =(t+1)\cdot\mathbb{E}[M_{r,d}]+\mathbb{E}[M_{r,d}]^2\leq(2t+1)\cdot\mathbb{E}[M_{r,d}].
\end{align*}
The first inequality uses the bound on $\mathbb{E}[M_S^2]$ and adds the nonnegative terms with $S=S'$ to the cross terms. The last inequality uses $\mathbb{E}[M_{r,d}]\leq t$.
The variable $M_{r,d}$ is independent of activation. Multiplying the last bound by $y_r/d$ therefore gives
\begin{equation}\label{eq:rbsc-group-second-moment}
 \mathbb{E}[W_{r,d}^2]\leq(2t+1)\cdot\mathbb{E}[W_{r,d}].
\end{equation}
Moreover, by linearity of expectation, independence of $A_{r,d}$ and $M_{r,d}$, and the first-moment calculation above,
\begingroup\small
\begin{align*}
 \mathbb{E}[W]
 &=\sum_{r,d}\Pr[A_{r,d}]\cdot\mathbb{E}[M_{r,d}]
 =\sum_{r,d}\frac1d\cdot\sum_{S\in\cS_{r,d}}
       \sum_{u\in\cB_S\cap\cB'}a_{u,S}
 =\sum_d\sum_{S\in\cS_d}\frac1d\cdot
       \br{\sum_{r\in\cR'_S\colon y_r>0}1}\cdot\sum_{u\in\cB_S\cap\cB'}a_{u,S}\\
 &=\sum_d\sum_{S\in\cS_d}\frac{\spr{\cR'_S}}d\cdot
       \sum_{u\in\cB_S\cap\cB'}a_{u,S}
 \leq2\cdot\sum_{u\in\cB'}\sum_{S\in\cS\colon u\in\cB_S}a_{u,S}
 =2\cdot\sum_{u\in\cB'}z_u\leq2\cdot t.
\end{align*}
\endgroup
To justify the fourth equality, fix $S\in\cS_d$. If $y_r>0$ for every $r\in\cR'_S$, then the number of terms in the preceding sum over $r$ is $\spr{\cR'_S}$. Otherwise, some $r\in\cR'_S$ has $y_r=0$, and~\eqref{eq:rbsc-lp-marginal} and~\eqref{eq:rbsc-lp-red} imply $0\leq a_{u,S}\leq x_S\leq y_r=0$ for every $u\in\cB_S$. The assignment sum for $S$ therefore vanishes, so both terms are zero. The first inequality uses $\spr{\cR'_S}/d<2$ and extends the sum to all sets using nonnegativity of the assignment variables. The fifth equality follows from~\eqref{eq:rbsc-lp-assign}, and the second inequality follows from~\eqref{eq:rbsc-lp-budget} and $z_u\geq0$. Finally, independence of the rounding choices for distinct group indices and~\eqref{eq:rbsc-group-second-moment} imply
\[
 \mathbb{E}[W^2]
 =\sum_{r,d}\mathbb{E}[W_{r,d}^2]
   +\sum_{r,d}\sum_{\substack{r',d'\\(r',d')\ne(r,d)}}\mathbb{E}[W_{r,d}]\cdot\mathbb{E}[W_{r',d'}]
 \leq(2t+1)\cdot\mathbb{E}[W]+\br{\mathbb{E}[W]}^2
 \leq8\cdot t^2+2\cdot t\leq10\cdot t^2,
\]
which gives the proof of the claim.
\end{proof}

To find an outcome with both sufficient coverage and small red cost, we define a \emph{potential}
\begin{equation}\label{eq:rbsc-potential}
 \Phi=M-\frac{W^2}{72\cdot t}-\frac{7\cdot t\cdot R}{360\cdot K\cdot \gamma}-\frac{t}{8}
\end{equation}
to be an affine combination of the random variables $M$, $W^2$, and $R$. The term involving $M$ rewards blue coverage, while the term involving $R$ penalizes red cost. Without the quadratic penalty, a large coverage reward could offset an arbitrarily large red cost. Since $W\geq M$, the quadratic penalty grows faster than the coverage reward and prevents this from happening. At the same time, our bound on the second moment of $W$ ensures that the penalty is small enough on average to keep the expected potential nonnegative. An outcome with nonnegative potential then guarantees both sufficient coverage and a small red cost.
\begin{claim}\label{claim:rbsc-positive-potential}
We have $\mathbb{E}[\Phi]\geq0$. Every output $\cC$ obtained from an outcome with $\Phi\geq0$ satisfies
\[
 \spr{\cov\br{\cC,\cB}}\geq t/8
 \qquad\text{and}\qquad
 \spr{\cov\br{\cC,\cR}}<920\cdot K\cdot \gamma.
\]
In particular, for the correct guess $K=\OPT\br{\cI}$, the red cost is smaller than $920\cdot\OPT\br{\cI}\cdot \gamma$.
\end{claim}
\begin{proof}
Claims~\ref{claim:rbsc-red-cost}, \ref{claim:rbsc-blue-coverage}, and~\ref{claim:rbsc-second-moment} give
\[
 \mathbb{E}[\Phi]
 >\frac{2\cdot t}{5}-\frac{10\cdot t^2}{72\cdot t}-\frac{7\cdot t\cdot7\cdot K\cdot \gamma}{360\cdot K\cdot \gamma}-\frac{t}{8}
 =0.
\]
If $\Phi\geq0$, all three subtracted terms are nonnegative, so $M\geq t/8$. Since $\cM\subseteq\cov\br{\cC}$, we have $\spr{\cov\br{\cC,\cB}}\geq M\geq t/8$. Furthermore, $W\geq M$ implies
\[
 \frac{7\cdot t\cdot R}{360\cdot K\cdot \gamma}
 \leq M-\frac{W^2}{72\cdot t}-\frac{t}{8}
 \leq M-\frac{M^2}{72\cdot t}-\frac{t}{8}
 =18\cdot t-\frac{(M-36\cdot t)^2}{72\cdot t}-\frac{t}{8}
 \leq\frac{143\cdot t}{8}.
\]
It follows that $\spr{\cov\br{\cC,\cR}}\leq R\leq\frac{6435}{7}\cdot K\cdot \gamma<920\cdot K\cdot \gamma$.
\end{proof}

Using the method of conditional expectations applied to the potential $\Phi$ in a relatively straightforward manner allows us to derandomize our procedure as follows.

\begin{claim}\label{claim:rbsc-conditional-expectations}
An outcome with $\Phi\geq0$ can be found deterministically in polynomial time.
\end{claim}
\begin{proof}
In order to obtain a deterministic algorithm, instead of drawing the coins randomly, we process the groups in an arbitrary order, first fixing their activation, and then, for an active group, fixing each occurrence's trial and, if it succeeds, its marking coins. At each step we choose a possible value maximizing the conditional expectation of $\Phi$.
Fix a history of exposed coins, and let $\mathcal{E}$ be the event that they take their recorded values. Since $\mathcal{E}$ is an intersection of individual coin outcomes, all unfixed coins remain independent. The conditional probability that a coin succeeds is its original probability if it is unfixed, and $0$ or $1$ otherwise.

For $S\in\cS_{r,d}$ and $u\in\cB_S\cap\cB'$, independence conditional on $\mathcal{E}$ gives
\[
 \Pr[u\in\cM_{r,d,S}\mid\mathcal{E}]
 =\Pr[T_{r,d,S}\mid\mathcal{E}]\cdot\Pr[Z_{r,d,S,u}=1\mid\mathcal{E}].
\]
For $u\in\cB'$ and $q\in\cR\setminus\cR_+$, the probabilities that the group misses a blue or red element are
\[
 \begin{aligned}
 \Pr[u\notin\cM_{r,d}\mid\mathcal{E}]
 &=1-\Pr[A_{r,d}\mid\mathcal{E}]+\Pr[A_{r,d}\mid\mathcal{E}]\cdot\prod_{S\in\cS_{r,d}\colon u\in\cB_S}\br{1-\Pr[u\in\cM_{r,d,S}\mid\mathcal{E}]},\\
 \Pr[q\notin\cR_{r,d}\mid\mathcal{E}]
 &=1-\Pr[A_{r,d}\mid\mathcal{E}]+\Pr[A_{r,d}\mid\mathcal{E}]\cdot\prod_{S\in\cS_{r,d}\colon q\in\cR'_S}\br{1-\Pr[T_{r,d,S}\mid\mathcal{E}]}.
 \end{aligned}
\]
Independence of distinct groups and linearity of expectation then give
\begingroup\small
\begin{equation}\label{eq:rbsc-conditional-coverage-cost}
 \mathbb{E}[M\mid\mathcal{E}]
 =\sum_{u\in\cB'}\br{1-\prod_{r,d}\Pr[u\notin\cM_{r,d}\mid\mathcal{E}]},\quad
 \mathbb{E}[R\mid\mathcal{E}]
 =\spr{\cR_+}+\sum_{q\in\cR\setminus \cR_+}\br{1-\prod_{r,d}\Pr[q\notin\cR_{r,d}\mid\mathcal{E}]}.
\end{equation}
\endgroup
For the second moment, fix a group $\cS_{r,d}$ and use the variables $N_S$ and $M_S$ defined in the proof of Claim~\ref{claim:rbsc-second-moment}. Conditional on $\mathcal{E}$, the marking coins and trial remain independent, so
\[
 \begin{gathered}
 \mathbb{E}[N_S\mid\mathcal{E}]=\sum_{u\in\cB_S\cap\cB'}\Pr[Z_{r,d,S,u}=1\mid\mathcal{E}],\qquad
 \mathbb{E}[M_S\mid\mathcal{E}]=\Pr[T_{r,d,S}\mid\mathcal{E}]\cdot\mathbb{E}[N_S\mid\mathcal{E}],\\
 \mathbb{E}[M_S^2\mid\mathcal{E}]=\Pr[T_{r,d,S}\mid\mathcal{E}]\cdot\br{\mathbb{E}[N_S\mid\mathcal{E}]^2+\sum_{u\in\cB_S\cap\cB'}\Pr[Z_{r,d,S,u}=1\mid\mathcal{E}]\cdot\br{1-\Pr[Z_{r,d,S,u}=1\mid\mathcal{E}]}}.
 \end{gathered}
\]
The variables $M_S$ remain independent across occurrences, and $M_{r,d}$ remains independent of activation. Thus
\[
 \begin{gathered}
 \mathbb{E}[M_{r,d}\mid\mathcal{E}]=\sum_{S\in\cS_{r,d}}\mathbb{E}[M_S\mid\mathcal{E}],\qquad
 \mathbb{E}[M_{r,d}^2\mid\mathcal{E}]=\mathbb{E}[M_{r,d}\mid\mathcal{E}]^2+\sum_{S\in\cS_{r,d}}\br{\mathbb{E}[M_S^2\mid\mathcal{E}]-\mathbb{E}[M_S\mid\mathcal{E}]^2},\\
 \mathbb{E}[W_{r,d}\mid\mathcal{E}]=\Pr[A_{r,d}\mid\mathcal{E}]\cdot\mathbb{E}[M_{r,d}\mid\mathcal{E}],\qquad
 \mathbb{E}[W_{r,d}^2\mid\mathcal{E}]=\Pr[A_{r,d}\mid\mathcal{E}]\cdot\mathbb{E}[M_{r,d}^2\mid\mathcal{E}].
 \end{gathered}
\]
Conditional on $\mathcal{E}$, the unfixed coins remain independent, and each $W_{r,d}$ depends only on coins belonging to its own group. Hence the variables $W_{r,d}$ are conditionally independent, and
\begin{equation}\label{eq:rbsc-conditional-second-moment}
 \mathbb{E}[W^2\mid\mathcal{E}]
 =\br{\sum_{r,d}\mathbb{E}[W_{r,d}\mid\mathcal{E}]}^2
  +\sum_{r,d}\br{\mathbb{E}[W_{r,d}^2\mid\mathcal{E}]-\mathbb{E}[W_{r,d}\mid\mathcal{E}]^2}.
\end{equation}
Substituting~\eqref{eq:rbsc-conditional-coverage-cost} and~\eqref{eq:rbsc-conditional-second-moment} into~\eqref{eq:rbsc-potential} computes $\mathbb{E}[\Phi\mid\mathcal{E}]$ in polynomial time.
% The same formulas compute the conditional expectation for each possible value of the next coin.
By averaging, one such value does not decrease the conditional expectation. There are polynomially many coins, so fixing them as above preserves the nonnegative initial expectation from Claim~\ref{claim:rbsc-positive-potential} in polynomial time.
\end{proof}

For the smallest feasible guess $K$, return $\cC=\cS_0$ if $\spr{\cov\br{\cS_0,\cB}}\geq t/8$. Otherwise, use Claim~\ref{claim:rbsc-conditional-expectations} to obtain a solution $\cC$ with a nonnegative potential. In both cases, $\spr{\cov\br{\cC,\cB}}\geq t/8$ and $\spr{\cov\br{\cC,\cR}}\leq920\cdot K\cdot \gamma$. Since $K=\OPT\br{\cI}$ is feasible by Claim~\ref{claim:rbsc-lp-validity}, the smallest feasible guess satisfies $K\leq\OPT\br{\cI}$. Hence, $\spr{\cov\br{\cC,\cR}}\leq920\cdot\OPT\br{\cI}\cdot \gamma = \bigo\br{m^{1/3}}\cdot \OPT\br{\cI}$ as promised.
\end{proof}

\subsection{\texorpdfstring{$\ProblemPCSC$}{PCSC} via Partial Red-Blue Set Cover}
\label{subsection:PCSC}
We now reduce $\ProblemPCSC$ to $\ProblemPRBSC$.
Given a precedence-constrained set covering instance $(\cU,\cS,\preceq,f)$, create a red element $r_S$ for each $S\in\cS$ and a blue element $b_u$ for each $u\in\cU$. For each $S\in\cS$, let $\cS_S=\closure{S}$ and create a red-blue set $A_S=\brc{r_T\colon T\in \cS_S}\cup\brc{b_u\colon u\in\cov\br{\cS_S}}$.
The resulting red-blue instance has $m$ sets and $n$ blue elements. For any selected subfamily $\mathcal{J}\subseteq\cS$, define a precedence-closed family $\cC_{\mathcal{J}}=\bigcup_{S\in\mathcal{J}}\cS_S$.
The red elements covered by the selected red-blue sets are in $\brc{r_T\colon T\in\cC_{\mathcal{J}}}$, and the blue elements covered are in $\brc{b_u\colon u\in\cov\br{\cC_{\mathcal{J}}}}$.
Thus, the red cost and blue coverage equal the size and coverage of $\cC_{\mathcal{J}}$, respectively.
Conversely, for any precedence-closed family $\cC\subseteq\cS$, selecting $A_S$ for every $S\in\cC$ covers the red elements corresponding to $\cC$ and the blue elements corresponding to $\cov\br{\cC}$, and no others: each $\cS_S\subseteq\cC$, and every $S\in\cC$ belongs to $\cS_S$. Hence, the reduction preserves cost and coverage in both directions.

\begin{theorem}\label{thm:PCSC-RBSC}
There is a polynomial-time $\br{\bigo\br{m^{1/3}},8}$-approximation algorithm for $\ProblemPCSC$.
\end{theorem}
\begin{proof}
Apply the reduction above with $t=\cl{f\cdot n}$, followed by Theorem~\ref{thm:partial-red-blue-bicriteria}. Since the reduction preserves cost and coverage, the theorem follows.
\end{proof}

\begin{corollary}\label{cor:bicriteria-decision-trees}
There is an $\bigo\br{m^{1/3}\cdot\log n}$-approximation algorithm for $\ProblemPCWCDT$ running in polynomial time.
\end{corollary}
\begin{proof}
Combine Theorem~\ref{thm:PCSC-RBSC} with Theorem~\ref{thm:PCSC-to-PCWCDT}.
\end{proof}


\subsection{\texorpdfstring{$\ProblemPCMSSC$ via $\ProblemBPCSC$}{PCMSSC via B-PCSC}} \label{subsection:PCMSSC}
To solve $\ProblemPCMSSC$, we first introduce a budgeted variant of precedence-constrained set cover.
\begin{problem}[\textup{\textsc{Budgeted Precedence-Constrained Set Cover}} ($\ProblemBPCSC$)] \label{problem:BPCSC}
Given a universe $\cU$ of items, a family $\cS$ of subsets of $\cU$, a precedence relation $(\cS,\preceq)$ and a budget $\budget>0$, find a precedence-closed set $\cC\subseteq\cS$ that maximises $\spr{\cov\br{\cC}}$ and satisfies $\spr{\cC}\leq \budget$.
\end{problem}

An approximation for $\ProblemPCSC$ gives the same bicriteria guarantee for $\ProblemBPCSC$ by guessing the fraction of elements covered by an optimal budgeted solution.
\begin{theorem}\label{thm:PCSC-to-BPCSC}\label{thm:BPCSC}
If there is a polynomial-time $\br{\alpha,\beta}$-approximation algorithm for $\ProblemPCSC$, then there is a polynomial-time $\br{\alpha,\beta}$-approximation algorithm for $\ProblemBPCSC$. In particular, $\ProblemBPCSC$ admits a polynomial-time $\br{\bigo\br{m^{1/3}},8}$-approximation algorithm.
\end{theorem}
\begin{proof}
For each integer $t\in[\spr{\cov\br{\cS}}]$, run the $\ProblemPCSC$ algorithm with target fraction $f=t/n$. Among the resulting families of size at most $\alpha\cdot\budget$ and the empty family, return one with maximum coverage.
Let $\cCopt$ be an optimal solution to $\ProblemBPCSC$ and put $t^*=\spr{\cov\br{\cCopt}}$. If $t^*>0$, then $\cCopt$ is feasible for $\ProblemPCSC$ with target fraction $f^*=t^*/n$, so the corresponding optimum is at most $\budget$. For the guess $t=t^*$, the algorithm therefore obtains a family of size at most $\alpha\cdot\budget$ covering at least $t^*/\beta$ elements. The stated bicriteria guarantee follows from Theorem~\ref{thm:PCSC-RBSC}.
\end{proof}

We show how to use an $(\alpha,\beta)$-approximation algorithm for $\ProblemBPCSC$ to obtain an $\br{\bigo\br{\alpha\cdot\beta^3}, \beta}$-approximation for $\ProblemPCMSSC$.  Our approach is reminiscent of the ideas used for solving \textup{\textsc{Minimum Latency TSP}}~\cite{PathsTreesMinimumLatencyTours,KTravelingRepairmenProblem} and \textup{\textsc{Adaptive TSP}}~\cite{ApproxAlgsForOptDTsAndAdapTSPProblems}. The overarching theme of this connection is to treat appending sets to a covering sequence as a kind of tour on the Hasse diagram of the precedence POSET, and the cost of covering an element as a latency of `visiting' that element. This intuition, however, is not exact and is not used directly in the procedure or its analysis.

Let $\alpha,\beta\geq 1$ and $a=\frac{3\cdot\beta+1}{3\cdot\beta-1}$, so $1<a\leq 2$.
% Denote by $\cCopt$ any optimal solution to $\ProblemPCMSSC$ on the input $(\cU, \cS, \preceq, f)$.
For an optimal solution $\cCopt$ to $\ProblemPCMSSC$ with $\spr{\cCopt}>0$, let $l^*=\max\brc{1,\cl{\log_a\spr{\cCopt}}}$.
So, $a^{l^*-1}\leq\spr{\cCopt}\leq a^{l^*}$.
We start with pseudocode for $\ProblemPCMSSC$ in Algorithm~\ref{alg:fPCMSSC}. The algorithm first guesses the value of $l^*$. Then, $\ProblemBPCSC$ is used with geometrically growing budgets
\input{pseudocodes/f_PCMSSC.tex}
Each family returned by $\ProblemBPCSC$ is precedence-closed, so the family $P_l$ of sets already appended is also precedence-closed. When a family is appended, only sets outside $P_l$ are used and they are ordered by $\operatorname{ord}$. Every predecessor of a new set is therefore either already in the sequence or earlier in the appended block. The same holds for the padding and the final family, so every sequence considered by the algorithm is precedence-closed and contains no repeated sets. In the analysis, $\pi_l$ denotes the padded prefix before the final family is appended.
We start with a lemma that relates the coverage times and the lengths of $\pi_{l^*}$ and $\cCopt$.
    \begin{lemma}\label{lem:pi-spread-cost}
        We have that $\spr{\pi_{l^*}} = \bigo\br{\alpha \cdot \beta}\cdot \spr{\cCopt}$, and if $\spr{\pi_{l^*}}<m$, then $\spr{\pi_{l^*}}=\Omega\br{\alpha\cdot\beta}\cdot\spr{\cCopt}$. Moreover, $\coverageTime\br{\pi_{l^*}} = \bigo\br{\alpha\cdot\beta^3}\cdot \coverageTime\br{\cCopt}$.
    \end{lemma}
    \begin{proof}
        We observe that since we used an $\br{\alpha, \beta}$-approximation algorithm for $\ProblemBPCSC$ in each iteration of the inner for loop, we have that for each $1\leq i\leq l^*$, $\spr{\tau_{l^*}^i} \leq \alpha \cdot a^{i+1}$. Thus, the length after padding satisfies:
        \begin{equation} \label{eq:pi}
        \spr{\pi_{l^*}} \leq \max\brc{\alpha\cdot \sum_{i=1}^{l^*} a^{i+1},\min\brc{m,\cl{\alpha\cdot\beta\cdot a^{l^*}}}}\leq \frac{\alpha\cdot a^{l^*+3}}{a-1}\leq \frac{\alpha\cdot a^{4}}{a-1}\cdot\spr{\cCopt}=\bigo\br{\alpha\cdot\beta}\cdot\spr{\cCopt}.
        \end{equation}
        Here we used $\cl{x}\leq x+1$, $\alpha\cdot a^{l^*}\geq1$, $\beta\leq1/(a-1)$, and $a^{l^*}\leq a\cdot\spr{\cCopt}$. If $\spr{\pi_{l^*}}<m$, the padding target is not capped by $m$, and hence $\spr{\pi_{l^*}}\geq\alpha\cdot\beta\cdot a^{l^*}\geq\alpha\cdot\beta\cdot\spr{\cCopt}$. If $\spr{\pi_{l^*}}=m$, then the prefix contains every set in $\cS$. This case will be handled after the latency bound.

        For every $i\in [l^*]$, let $\ell_i=\min\brc{\spr{\cCopt},\fl{a^i}}$ and $n_i^*=\spr{\cov\br{\pref{\cCopt}{\ell_i}}}$, so $n_{l^*}^* = \spr{\cov\br{\cCopt}}$. Let $n_i=\spr{\cov\br{\bigcup_{j=1}^{i}\tau_{l^*}^j}}$ and $n_0=n_0^*=0$. We have:
        \begin{align*}
            &\coverageTime\br{\pi_{l^*}}\leq \sum_{i=1}^{l^*}\br{n_i-n_{i-1}}\cdot \sum_{j=1}^{i}\alpha\cdot a^{j+1} + \br{n-n_{l^*}}\cdot \spr{\pi_{l^*}}
            \\&\leq
            \sum_{i=1}^{l^*}\br{n_i-n_{i-1}}\cdot \frac{\alpha\cdot a^{i+2}}{a-1} + \br{n-n_{l^*}}\cdot \spr{\pi_{l^*}}
            \\&=
            \sum_{i=1}^{l^*}\br{\br{n-n_{i-1}}-\br{n-n_i}}\cdot \frac{\alpha\cdot a^{i+2}}{a-1} + \br{n-n_{l^*}}\cdot \spr{\pi_{l^*}} \leq
            \sum_{i=0}^{l^*}\br{n-n_i}\cdot \frac{\alpha\cdot a^{i+3}}{a-1} = Q.
        \end{align*}
        Here $Q$ denotes the last sum, and the last inequality follows from \eqref{eq:pi}.
        Moreover, we have that:
        \begin{align*}
            &2\cdot\coverageTime\br{\cCopt}\geq a^{l^*-1}\cdot\br{n-n_{l^*-1}^*}+ \sum_{i=1}^{l^*-1}a^{i-1}\cdot\br{n_i^*-n_{i-1}^*}+\br{n-n_{l^*}^*}\cdot\spr{\cCopt}
            \\&\geq
        a^{l^*-1}\cdot\br{n-n_{l^*-1}^*}+ \sum_{i=1}^{l^*-1}a^{i-1}\cdot\br{\br{n-n_{i-1}^*}-\br{n-n_{i}^*}}+\br{n-n_{l^*}^*}\cdot a^{l^*-1}
            \\&=
        \br{n-n_0^*}+\sum_{i=1}^{l^*-1}\br{a^{i}-a^{i-1}}\cdot\br{n-n_{i}^*}+\br{n-n_{l^*}^*}\cdot a^{l^*-1} \geq
        \br{1-\frac{1}{a}}\cdot\sum_{i=0}^{l^*}a^{i}\cdot \br{n-n_i^*}.
        \end{align*}

        Consider any iteration $i\in \brc{1,\dots,l^*}$ of the inner for loop. Let $\cR_{i-1}=\cU\setminus\cov\br{\bigcup_{j=1}^{i-1}\tau_{l^*}^j}$ be the algorithmic residual at the beginning of iteration $i$. The family $\pref{\cCopt}{\ell_i}$ is precedence-closed, has size $\ell_i\leq a^i\leq a^{i+1}$, and is therefore feasible for the $i$-th $\ProblemBPCSC$ instance. Since it covers $n_i^*$ elements in total and the previously computed solutions cover $n_{i-1}$ elements, it covers at least $n_i^*-n_{i-1}$ elements in $\cR_{i-1}$.
        Thus, $n_i^*-n_{i-1}$ lower bounds the optimum residual coverage in this iteration.
        Hence, by the $\br{\alpha,\beta}$-approximation guarantee, $n_i-n_{i-1}\geq\frac{n_i^*-n_{i-1}}{\beta}$. Rearranging gives $n-n_i\leq\frac{\beta-1}{\beta}\cdot\br{n-n_{i-1}}+\frac{1}{\beta}\cdot\br{n-n_i^*}$.
        Thus,
        \begin{align*}
            &\frac{\br{a-1}\cdot Q}{\alpha}
            =
            \sum_{i=0}^{l^*}a^{i+3}\cdot \br{n-n_i} \leq
            a^3\cdot n + \frac{1}{\beta}\cdot\sum_{i=1}^{l^*}a^{i+3}\cdot \br{n-n_i^*} + \frac{\beta-1}{\beta}\cdot\sum_{i=1}^{l^*}a^{i+3}\cdot \br{n-n_{i-1}}
            \\&\leq
            \frac{2\cdot a^4}{a-1}\cdot\coverageTime\br{\cCopt}+\frac{\beta-1}{\beta}\cdot\sum_{i=1}^{l^*}a^{i+3}\cdot \br{n-n_{i-1}}
            \\&=
            \frac{2\cdot a^4}{a-1}\cdot\coverageTime\br{\cCopt}+\frac{\br{\beta-1}\cdot a}{\beta}\cdot\sum_{i=0}^{l^*-1}a^{i+3}\cdot \br{n-n_{i}} \leq
            \frac{2\cdot a^4}{a-1}\cdot\coverageTime\br{\cCopt}+\frac{\br{\beta-1}\cdot a}{\beta}\cdot\br{a-1}\cdot\frac{Q}{\alpha}.
        \end{align*}
        Rearranging and substituting $a=(3\cdot\beta+1)/(3\cdot\beta-1)$ gives
        \begin{align*}
            \frac{Q}{\alpha\cdot\coverageTime\br{\cCopt}}
            &\leq \frac{2\cdot a^4}{(a-1)^2\cdot\br{1-\frac{(\beta-1)\cdot a}{\beta}}}
            =\frac{\beta\cdot(3\cdot\beta+1)^4}{2\cdot(\beta+1)\cdot(3\cdot\beta-1)}
            =\bigo\br{\beta^3}.
        \end{align*}
        Here $a-1=2/(3\cdot\beta-1)$ and $1-(\beta-1)\cdot a/\beta=(\beta+1)/(\beta\cdot(3\cdot\beta-1))$.
        The lemma follows.
    \end{proof}

    If $\spr{\pi_{l^*}}=m$, then $\pi_{l^*}$ contains every set in $\cS$, and thus covers every element covered by $\cCopt$. The candidate is feasible and its latency is bounded by Lemma~\ref{lem:pi-spread-cost}. Otherwise, let $P_{l^*}$ be the family of sets in $\pi_{l^*}$, let $\cR_{l^*}=\cU\setminus\cov\br{P_{l^*}}$, and let $r=\spr{\cov\br{P_{l^*}}}$.
    \begin{lemma}\label{lem:sigma-coverage}
        We have that $\spr{\cov\br{\sigma_{l^*},\cR_{l^*}}} \geq \frac{\max\brc{0,f\cdot n-r}}{\beta}$ and $\spr{\sigma_{l^*}} = \bigo\br{\alpha}\cdot \spr{\cCopt}$.
    \end{lemma}
    \begin{proof}
        Since $\cCopt$ satisfies $\spr{\cCopt}\leq a^{l^*}$, it is feasible for the final $\ProblemBPCSC$ instance on residual universe $\cR_{l^*}$. It covers at least $\max\brc{0,f\cdot n-r}$ elements of that residual. The approximation guarantee gives the coverage bound, while $\spr{\sigma_{l^*}}\leq\alpha\cdot a^{l^*}\leq\alpha\cdot a\cdot\spr{\cCopt}=\bigo\br{\alpha}\cdot\spr{\cCopt}$ gives the size bound.
    \end{proof}
    Let $\pi_{l^*}^+=\pi_{l^*}\circ\operatorname{ord}\br{\sigma_{l^*}\setminus P_{l^*}}$ denote the sequence obtained by appending only the final solution's new sets.
    \begin{lemma}\label{lem:combined-coverage-cost}
        We have that $\spr{\cov\br{\pi_{l^*}^+}}\geq \frac{f\cdot n}{\beta}$ and $\coverageTime\br{\pi_{l^*}^+} = \bigo\br{\alpha\cdot\beta^3}\cdot \coverageTime\br{\cCopt}$.
    \end{lemma}
    \begin{proof}
        By Lemma~\ref{lem:sigma-coverage}, the prefix and the final solution together cover at least $r+\frac{\max\brc{0,f\cdot n-r}}{\beta}\geq\frac{f\cdot n}{\beta}$ elements. Here the residual coverage from $\sigma_{l^*}$ is disjoint from the prefix coverage by definition of $\cR_{l^*}$.
        For each element $u\in\cU$, let $t_u$ denote the index of the first set in $\pi_{l^*}$ that covers $u$ if $u$ is covered by $\pi_{l^*}$, otherwise, set $t_u=\spr{\pi_{l^*}}$.
        Lemma \ref{lem:pi-spread-cost} implies that 
        \[
        \coverageTime\br{\pi_{l^*}}=\sum_{u\in\cU} t_u = \bigo\br{\alpha\cdot\beta^3}\cdot \coverageTime\br{\cCopt}.
        \]
        Observe that for each $u\in\cov\br{\pi_{l^*}}$, its coverage time in $\pi_{l^*}^+$ is also $t_u$.
        For each $u\notin\cov\br{\pi_{l^*}}$, we have $t_u=\spr{\pi_{l^*}}$. Since this case has $\spr{\pi_{l^*}}<m$, the padding guarantee gives $\spr{\pi_{l^*}}\geq\alpha\cdot\beta\cdot a^{l^*}$. The appended suffix contains at most $\spr{\sigma_{l^*}}\leq\alpha\cdot a^{l^*}\leq\spr{\pi_{l^*}}/\beta$ new sets. Thus the coverage time of $u$ in $\pi_{l^*}^+$ is at most $(1+1/\beta)\cdot t_u$.
        The explicit bound in the proof of Lemma~\ref{lem:pi-spread-cost} therefore gives
        \[
        \coverageTime\br{\pi_{l^*}^+}
        \leq\br{1+\frac{1}{\beta}}\cdot\coverageTime\br{\pi_{l^*}}
        \leq\frac{\alpha\cdot(3\cdot\beta+1)^4}{2\cdot(3\cdot\beta-1)}\cdot\coverageTime\br{\cCopt}
        =\bigo\br{\alpha\cdot\beta^3}\cdot\coverageTime\br{\cCopt}.
        \]
    \end{proof}
Since $\log a=\Theta(1/\beta)$, Algorithm~\ref{alg:fPCMSSC} makes $\bigo((1+\beta\log(m+1))^2)$ calls to the budgeted algorithm. Its running time is therefore polynomial whenever $\beta$ is polynomially bounded in the input size. All applications below use constant $\beta$.
By the full-prefix case above and Lemma~\ref{lem:combined-coverage-cost}, the candidate for $l^*$ meets the coverage requirement and has the stated latency bound. Since Algorithm~\ref{alg:fPCMSSC} returns the minimum-latency candidate meeting that coverage requirement, the following theorem follows.
\begin{theorem}\label{thm:BPCSC-to-PCMSSC}
    \setlength{\emergencystretch}{2em}
    If there exists a polynomial-time bicriteria $(\alpha,\beta)$-approximation algorithm for $\ProblemBPCSC$, with $\beta$ polynomially bounded in the input size, then there exists a polynomial-time bicriteria $\br{\bigo\br{\alpha\cdot\beta^3}, \beta}$-approximation algorithm for $\ProblemPCMSSC$.
\end{theorem}
It should be noted that in all cases when we use Theorem~\ref{thm:BPCSC-to-PCMSSC} we will have $\beta=\bigo\br{1}$, thus yielding a $\br{\bigo\br{\alpha}, \bigo\br{1}}$-approximation algorithm for $\ProblemPCMSSC$.

\begin{corollary}\label{cor:PCACDT-general}
There is a deterministic polynomial-time $\bigo\br{m^{1/3}\cdot\log n}$-approximation algorithm for $\ProblemPCACDT$ under general precedence constraints.
\end{corollary}
\begin{proof}
Combine the budgeted guarantee in Theorem~\ref{thm:PCSC-to-BPCSC} with Theorems~\ref{thm:BPCSC-to-PCMSSC} and~\ref{thm:PCMSSC-to-PCACDT}.
\end{proof}

\subsection{Special cases of precedence constraints} \label{subsection:SpecialCases}
In \textup{\textsc{Budgeted Maximum Coverage}}, the input consists of a universe $\cU$, a family $\cS$ of subsets of $\cU$, each with a cost $c(S)\geq0$, and a budget $\budget>0$. The goal is to maximize $\spr{\cov\br{\cS'}}$ over subfamilies $\cS'\subseteq\cS$ with $\sum_{S\in\cS'}c(S)\leq\budget$.

\begin{theorem}[Khuller, Moss and Naor \cite{TheBudgetedMaximumCoverageProblem}]\label{thm:budgeted-maximum-coverage-source}
    There exists a polynomial-time $(1-1/e)$-approximation algorithm for \textup{\textsc{Budgeted Maximum Coverage}}.
\end{theorem}

\begin{theorem}\label{thm:BPCSC-inforest}
    \setlength{\emergencystretch}{2em}
    There exist polynomial-time bicriteria $\br{1, \frac{e}{e-1}}$-approximation algorithms for $\ProblemBPCSC$ and $\ProblemPCSC$ when the precedence constraints form an inforest.
\end{theorem}
\begin{proof}
    We reduce $\ProblemBPCSC$ to \textup{\textsc{Budgeted Maximum Coverage}} and apply Theorem~\ref{thm:budgeted-maximum-coverage-source}.

    Given a $\ProblemBPCSC$ instance $(\cU,\cS,\preceq,\budget)$, construct a \textup{\textsc{Budgeted Maximum Coverage}} instance with universe $\cU$ and the budget $\budget$. For each $S\in\cS$, create a set $C_S=\cov\br{\closure{S}}$ with cost $c(C_S)=\spr{\closure{S}}$, and put $\cS'=\brc{C_S\colon S\in\cS}$.
    For any selected family $\cA\subseteq\cS'$, define $\cC=\bigcup_{C_S\in\cA}\closure{S}$. This family is precedence-closed and satisfies
    \[
        \cov\br{\cC}=\cov\br{\cA}
        \qquad\text{and}\qquad
        \spr{\cC}\leq\sum_{T\in\cA}c(T).
    \]
    In an inforest, intersecting closures are nested: for $S,W\in\cS$, if $\closure{S}\cap\closure{W}\neq\emptyset$, then $S\preceq W$ or $W\preceq S$. For any precedence-closed family $\cC\subseteq\cS$, let $\mathcal{M}$ be the family of its $\preceq$-maximal members and put $\cA_{\cC}=\brc{C_S\colon S\in\mathcal{M}}$. Then
    \[
        \cC=\biguplus_{S\in\mathcal{M}}\closure{S}
        \qquad\text{and}\qquad
        \cov\br{\cA_{\cC}}=\cov\br{\cC}.
    \]
    By the definition of the costs,
    \[
        \sum_{T\in\cA_{\cC}}c(T)
        =\sum_{S\in\mathcal{M}}\spr{\closure{S}}=\spr{\cC}.
    \]
    The maximum coverage attainable within the budget $\budget$ is therefore the same in both instances, which gives the $\ProblemBPCSC$ guarantee.

For $\ProblemPCSC$, run the same construction for every integer budget in $[m]$ and return a smallest family covering at least $(1-1/e)\cdot f\cdot n$ elements among the outputs and the empty family.
%  Let $\cCopt$ be an optimal $\ProblemPCSC$ solution. For $\spr{\cCopt}>0$, the budget $\spr{\cCopt}$ yields such a family of size at most $\spr{\cCopt}$.
\end{proof}

The inforest structure also allows us to give a stronger guarantee for $\ProblemPCMSSC$ using a greedy procedure (cf. Algorithm~\ref{alg:inforest-fPCMSSC}). The algorithm guesses the size of an optimal solution and repeatedly appends a set together with its predecessors not yet selected, choosing the most newly covered elements per added set among choices that add at most the guessed number of sets. In an inforest, every precedence-closed family decomposes into disjoint closures, which lets us compare this choice with any prefix of an optimal solution. Unlike the histogram-shrinkage argument for \textup{\textsc{Min-Sum Set Cover}}~\cite{ApproximatingMinSumSetCover}, which compares histograms associated with the greedy and optimal solutions, we use the optimal prefixes to bound coverage time directly. Our analysis accounts for elements left uncovered when the algorithm stops and handles the final iteration separately.

In the algorithm and its analysis, we identify a sequence with its family of sets when using set operations.
\input{pseudocodes/inforest_f_PCMSSC.tex}
\begin{corollary}\label{cor:BPCSC-inforest-to-PCMSSC}
    \setlength{\emergencystretch}{2em}
    There exists a polynomial-time bicriteria $\br{4,\frac{e}{e-1}}$-approximation algorithm for $\ProblemPCMSSC$ when the precedence constraints form an inforest.
\end{corollary}
\begin{proof}
Consider sequences constructed by Algorithm~\ref{alg:inforest-fPCMSSC} as candidate solutions. Each of them respects the precedence constraints: initially $\pi$ is empty, and for each selected $S\in\cS$, the family $\pi\cup D_S=\pi\cup\closure{S}$, where $D_S=\closure{S}\setminus\pi$, is precedence-closed whenever $\pi$ is and the elements of $D_S$ are appended in a topological order.
Let $\cCopt$ be an optimal solution and put $B^*=\spr{\cCopt}$. We analyze the guess $B=B^*$ and let $\ell$ be the number of iterations of its while loop. For each $i\in[\ell]$, let $\pi_i$ be the sequence at the start of iteration $i$ and put
\[
    r_i=n-\spr{\cov\br{\pi_i}},\qquad
    \delta_i=\spr{\cov\br{\cCopt}}-\spr{\cov\br{\pi_i}}.
\]
Let $\pi_{\ell+1}$ be the sequence after the loop. Fix $i\in[\ell]$. Since $\spr{\cov\br{\cCopt}}\geq f\cdot n$, we have
\[
    \delta_i>\spr{\cov\br{\cCopt}}-\br{1-1/e}\cdot f\cdot n\geq\spr{\cov\br{\cCopt}}/e>0.
\]
For each $S\in\cS$, let $D_{i,S}=\closure{S}\setminus\pi_i$, $c_{i,S}=\spr{D_{i,S}}$ and $g_{i,S}=\spr{\cov\br{D_{i,S},\cU\setminus\cov\br{\pi_i}}}$. As in the algorithm, let $\cA_i=\brc{S\in\cS\colon 1\leq c_{i,S}\leq B^*,\ g_{i,S}>0}$. For $j\in[B^*]$, let $\mathcal{M}_j$ be the family of $\preceq$-maximal members of $\pref{\cCopt}{j}$. The inforest property gives
\[
    \pref{\cCopt}{j}=\biguplus_{S\in\mathcal{M}_j}\closure{S},\qquad
    \sum_{S\in\mathcal{M}_j}c_{i,S}
    \leq\sum_{S\in\mathcal{M}_j}\spr{\closure{S}}
    =\spr{\pref{\cCopt}{j}}=j.
\]
\begin{claim}\label{claim:inforest-pcmssc-coverage}
In any iteration $i$, $\cA_i\neq\emptyset$. Moreover, $\pi_{\ell+1}\in\cQ$ satisfies $\spr{\cov\br{\pi_{\ell+1}}}\geq(1-1/e)\cdot f\cdot n$.
\end{claim}
\begin{proof}
For $j=B^*$, we have
\begin{align*}
    \sum_{S\in\mathcal{M}_{B^*}}g_{i,S}
    &=\sum_{S\in\mathcal{M}_{B^*}}\spr{\cov\br{D_{i,S},\cU\setminus\cov\br{\pi_i}}}
    \geq\spr{\bigcup_{S\in\mathcal{M}_{B^*}}\cov\br{D_{i,S},\cU\setminus\cov\br{\pi_i}}}\\
    &=\spr{\cov\br{\cCopt}\setminus\cov\br{\pi_i}}\geq\delta_i>0.
\end{align*}
Thus, there exists $S\in\mathcal{M}_{B^*}$ with $g_{i,S}>0$. For this set, $D_{i,S}\neq\emptyset$ and hence $1\leq c_{i,S}\leq B^*$, so $S\in\cA_i$. As $i\in[\ell]$ was arbitrary, each iteration appends at least one set, so the algorithm reaches the coverage threshold for this guess. Thus $\pi_{\ell+1}\in\cQ$ and $\spr{\cov\br{\pi_{\ell+1}}}\geq(1-1/e)\cdot f\cdot n$.
\end{proof}

For each $i\in[\ell]$, let $S_i$ be the set selected in iteration $i$, and denote for brevity $c_i=c_{i,S_i}$, $g_i=g_{i,S_i}$ and $\rho_i=g_i/c_i$. For the fixed iteration $i$, we have
\[
    \rho_i=\frac{g_i}{c_i}=\max_{S\in\cA_i}\frac{g_{i,S}}{c_{i,S}}.
\]

For $j\in\brc{0,\ldots,B^*}$, let $r_j^*=n-\spr{\cov\br{\pref{\cCopt}{j}}}$ be the number of elements not covered by the first $j$ sets of $\cCopt$, where $\pref{\cCopt}{0}$ is empty.
\begin{claim}\label{claim:inforest-pcmssc-density}
We have $\rho_i\geq\delta_i/B^*$. If $r_i>2\cdot\coverageTime\br{\cCopt}/B^*$, then $\rho_i\geq r_i^2/\br{2\cdot\coverageTime\br{\cCopt}}$.
\end{claim}
\begin{proof}
Since uncovered elements have coverage time $B^*$, the definition of coverage time gives
\[
    \coverageTime\br{\cCopt}=\sum_{j=0}^{B^*-1}r_j^*.
\]
For each $j\in[B^*]$ and $S\in\mathcal{M}_j$ with $g_{i,S}>0$, we have $S\in\cA_i$ and hence $g_{i,S}\leq\rho_i\cdot c_{i,S}$. The same inequality holds when $g_{i,S}=0$. Thus,
\begin{align*}
    r_i-r_j^*&=\spr{\cov\br{\pref{\cCopt}{j}}}-\spr{\cov\br{\pi_i}}
    \leq\spr{\cov\br{\pref{\cCopt}{j}}\setminus\cov\br{\pi_i}}\\
    &=\spr{\bigcup_{S\in\mathcal{M}_j}\cov\br{D_{i,S},\cU\setminus\cov\br{\pi_i}}}
    \leq\sum_{S\in\mathcal{M}_j}g_{i,S}
    \leq\rho_i\cdot\sum_{S\in\mathcal{M}_j}c_{i,S}\leq\rho_i\cdot j.
\end{align*}
Consequently, since $r_0^*=n\geq r_i$,
\[
    r_j^*\geq\max\brc{0,r_i-\rho_i\cdot j},\qquad
    \rho_i\geq\frac{\delta_i}{B^*}.
\]
The last inequality uses $j=B^*$ and $r_{B^*}^*=r_i-\delta_i$.
Suppose that $r_i>2\cdot \coverageTime\br{\cCopt}/B^*$. If $\rho_i<r_i/B^*$, then
\[
    \coverageTime\br{\cCopt}\geq\sum_{j=0}^{B^*-1}\br{r_i-\rho_i\cdot j}
    =r_i\cdot B^*-\frac{\rho_i\cdot B^*\cdot(B^*-1)}{2}
    \geq\frac{r_i\cdot(B^*+1)}{2}>\frac{r_i\cdot B^*}{2}>\coverageTime\br{\cCopt},
\]
a contradiction. Hence $r_i/\rho_i\leq B^*$. Put $h=\cl{r_i/\rho_i}$, so $1\leq h\leq B^*$ and $0\leq h-r_i/\rho_i<1$. Summing over the first $h$ terms gives
\begin{align*}
    \coverageTime\br{\cCopt}&\geq\sum_{j=0}^{h-1}\br{r_i-\rho_i\cdot j}
    =h\cdot r_i-\frac{\rho_i\cdot h\cdot(h-1)}{2}
    =\frac{\rho_i}{2}\cdot\br{\frac{2\cdot h\cdot r_i}{\rho_i}-h^2+h}\\
    &=\frac{\rho_i}{2}\cdot\br{\frac{r_i^2}{\rho_i^2}+h-\br{h-\frac{r_i}{\rho_i}}^2}
    =\frac{r_i^2}{2\cdot\rho_i}+\frac{\rho_i}{2}\cdot\br{h-\br{h-\frac{r_i}{\rho_i}}^2}
    \geq\frac{r_i^2}{2\cdot\rho_i}.
\end{align*}
The last inequality uses $h-(h-r_i/\rho_i)^2\geq h-1\geq0$. Therefore $\rho_i\geq\frac{r_i^2}{2\cdot\coverageTime\br{\cCopt}}$.
\end{proof}

\begin{claim}\label{claim:inforest-pcmssc-cost}
We have $\coverageTime\br{\pi_{\ell+1}}<4\cdot\coverageTime\br{\cCopt}$.
\end{claim}
\begin{proof}
Each element uncovered before iteration $i$ contributes at most $c_i$ to its coverage time during that iteration, and hence
\[
    \coverageTime\br{\pi_{\ell+1}}\leq\sum_{i=1}^{\ell} c_i\cdot r_i.
\]
Since $i\in[\ell]$ was arbitrary, Claim~\ref{claim:inforest-pcmssc-density} and $r_i-\delta_i=n-\spr{\cov\br{\cCopt}}\geq0$ give, for every $i\in[\ell]$:
\begin{enumerate}
    \item If $r_i>2\cdot\coverageTime\br{\cCopt}/B^*$, then
    \[
        c_i\cdot r_i=\frac{g_i\cdot r_i}{\rho_i}\leq2\cdot\coverageTime\br{\cCopt}\cdot\frac{g_i}{r_i}\leq2\cdot\coverageTime\br{\cCopt}\cdot\frac{g_i}{\delta_i}.
    \]
    \item If $r_i\leq2\cdot\coverageTime\br{\cCopt}/B^*$, then
    \[
        c_i\cdot r_i=\frac{g_i\cdot r_i}{\rho_i}\leq\frac{B^*\cdot g_i\cdot r_i}{\delta_i}\leq2\cdot\coverageTime\br{\cCopt}\cdot\frac{g_i}{\delta_i}.
    \]
\end{enumerate}
For $i\in[\ell-1]$, the loop condition gives $\delta_{i+1}>0$, and
\[
    \delta_i-\delta_{i+1}=\spr{\cov\br{\pi_{i+1}}}-\spr{\cov\br{\pi_i}}
    =\spr{\cov\br{D_{i,S_i},\cU\setminus\cov\br{\pi_i}}}=g_i.
\]
Hence,
\[
    \frac{g_i}{\delta_i}=\sum_{j=\delta_{i+1}+1}^{\delta_i}\frac{1}{\delta_i}
    \leq\sum_{j=\delta_{i+1}+1}^{\delta_i}\frac{1}{j}
    \leq\sum_{j=\delta_{i+1}+1}^{\delta_i}\ln\br{\frac{j}{j-1}}
    =\ln\br{\frac{\delta_i}{\delta_{i+1}}}.
\]
The second inequality uses $1/j\leq\ln\br{j/(j-1)}$ for $j\geq2$, which follows from $\ln x\leq x-1$ with $x=(j-1)/j$.
Since $\delta_1=\spr{\cov\br{\cCopt}}$ and $\delta_{\ell}>\delta_1/e$, summing and telescoping yields
\[
    \coverageTime\br{\pi_{\ell}}\leq\sum_{i=1}^{\ell-1}c_i\cdot r_i\leq2\cdot\coverageTime\br{\cCopt}\cdot\sum_{i=1}^{\ell-1}\frac{g_i}{\delta_i}
    \leq2\cdot\coverageTime\br{\cCopt}\cdot\sum_{i=1}^{\ell-1}\ln\br{\frac{\delta_i}{\delta_{i+1}}}
    =2\cdot\coverageTime\br{\cCopt}\cdot\ln\br{\frac{\delta_1}{\delta_{\ell}}}<2\cdot\coverageTime\br{\cCopt}.
\]
For the final iteration, Claim~\ref{claim:inforest-pcmssc-density} gives the following two cases:
\begin{enumerate}
    \item If $r_{\ell}>2\cdot\coverageTime\br{\cCopt}/B^*$, then $g_{\ell}\leq r_{\ell}$ gives
    \[
        c_{\ell}\cdot r_{\ell}\leq2\cdot\coverageTime\br{\cCopt}\cdot\frac{g_{\ell}}{r_{\ell}}\leq2\cdot\coverageTime\br{\cCopt}.
    \]
    \item If $r_{\ell}\leq2\cdot\coverageTime\br{\cCopt}/B^*$, then $c_{\ell}\leq B^*$ gives
    \[
        c_{\ell}\cdot r_{\ell}\leq B^*\cdot r_{\ell}\leq2\cdot\coverageTime\br{\cCopt}.
    \]
\end{enumerate}
Combining the bounds for the first $\ell-1$ iterations and the final iteration proves the claim.
\end{proof}

Let $\pi'$ be the sequence returned by the algorithm. By Claims~\ref{claim:inforest-pcmssc-coverage} and~\ref{claim:inforest-pcmssc-cost},
\[
    \coverageTime\br{\pi'}
    \leq\coverageTime\br{\pi_{\ell+1}}<4\cdot\coverageTime\br{\cCopt},\qquad
    \spr{\cov\br{\pi'}}\geq(1-1/e)\cdot f\cdot n.
\]

Each iteration that does not terminate the loop appends at least one new set. There are polynomially many guesses and at most $m$ such iterations per guess. Thus the algorithm runs in polynomial time and returns a candidate with the stated guarantees.
\end{proof}

\paraTitle{Steiner problems.}
Let $(V,d)$ denote a symmetric metric with distance function $d\colon V \times V \to \mathbb{R}_{\geq 0}$ satisfying the triangle inequality.
An \emph{$r$-tour} $\tau$ is a walk that starts and ends at $r \in V$. Denote by $\operatorname{len}\br{\tau}$ the length of tour $\tau$ with respect to $d$.
Let $\cX$ be a set of distinct \emph{groups}, where $X\subseteq V$ for each $X \in \cX$.
Let $w\colon \cX \to \mathbb{R}_{\geq 0}$ be a weight assignment to groups.
Let $V(\tau)$ be the set of vertices visited by $\tau$. The set of groups \emph{covered} by $\tau$ is $\cX_\tau=\brc{X\in\cX\colon X\cap V(\tau)\neq\emptyset}$.
% For a tour $\tau$ and a group $X \in \cX$, define $t_\tau(X)$ as the length of the shortest prefix of $\tau$ (with respect to $d$) that visits at least one vertex from $X$. If $X \notin \cX_\tau$, we set $t_\tau(X) = \spr{\tau}$.

\begin{problem}[\textup{\textsc{Group Steiner Orienteering}} ($\ProblemGSO$)] \label{problem:GSO}
Given a metric space $(V, d)$, a root vertex $r \in V$, a set of groups $\cX \subseteq 2^V$, a weight function $w\colon \cX \to \mathbb{R}_{\geq 0}$, and a budget $\budget > 0$, find an $r$-tour $\tau$ with $\operatorname{len}\br{\tau} \leq \budget$ that maximizes
$\sum_{X \in \cX_\tau} w(X)$.
\end{problem}
The following tree-metric guarantee follows from the algorithm and its analysis in Section~5.1 of \cite{ApproxAlgsForOptDTsAndAdapTSPProblems}: for a tree metric, the tree-embedding step is unnecessary, saving a logarithmic factor.
\begin{theorem}[Gupta, Nagarajan and Ravi \cite{ApproxAlgsForOptDTsAndAdapTSPProblems}, Section~5.1]\label{thm:gso-tree-source}
    Consider a $\ProblemGSO$ instance $\cI$ on a tree metric $(V,d)$ with a budget $\budget$.
    There exists a polynomial-time algorithm that finds an $r$-tour of length $\bigo\br{\log\spr{V}}\cdot\budget$ covering groups of total weight at least $\frac{1}{4}\OPT\br{\cI}$.
\end{theorem}

    There exist polynomial-time bicriteria $\br{\bigo\br{\log m}, 4}$-approximation algorithms for $\ProblemBPCSC$,  $\ProblemPCSC$ and $\ProblemPCMSSC$ when the precedence constraints form an outforest.

\begin{proof}
    We reduce $\ProblemBPCSC$ to $\ProblemGSO$ on tree metrics and apply Theorem~\ref{thm:gso-tree-source}. Let $(\cU,\cS,\preceq,\budget)$ be a $\ProblemBPCSC$ instance.
    Create a rooted tree with vertex set $V=\brc{r}\cup\brc{v_S\colon S\in\cS}$, where $r$ is a dummy root. For every minimal $S\in\cS$, connect $r$ to $v_S$ by an edge of length~$1/2$. For every nonminimal $W\in\cS$, let $S\in\cS$ be its unique immediate predecessor, and connect $v_S$ to $v_W$ by an edge of length~$1/2$. Thus, the path from $r$ to $v_W$ visits the representatives of all predecessors of $W$. For each $u\in\cU$, let $X_u=\brc{v_S\colon S\in\cS,\ u\in S}$. Set the GSO budget to $\budget$ and define its groups and their weights by
    \[
        \cX=\brc{X_u\colon u\in\cU,\ X_u\neq\emptyset},
        \qquad w(X)=\spr{\brc{u\in\cU\colon X_u=X}}\quad(X\in\cX).
    \]
    Here $\cX$ contains each distinct nonempty set $X_u$ once.

    Expand each edge of an $r$-tour in the tree metric along its tree path. The resulting walk $\tau$ has the same length and covers every group covered by the original tour. Let $\cC_\tau=\brc{S\in\cS\colon v_S\in V(\tau)}$. The root-to-vertex paths imply that $\cC_\tau$ is precedence-closed. The subtree formed by these paths has $\spr{\cC_\tau}$ edges, one for each visited representative. Since $\tau$ starts and ends at $r$, it traverses each edge in this subtree at least twice. As every edge has length~$1/2$, we have $\spr{\cC_\tau}\leq\operatorname{len}\br{\tau}$. Moreover,
    \[
        \sum_{X\in\cX_\tau}w(X)
        =\spr{\brc{u\in\cU\colon X_u\cap V(\tau)\neq\emptyset}}
        =\spr{\cov\br{\cC_\tau}}.
    \]

    Conversely, for any precedence-closed family $\cC\subseteq\cS$, the union of the root-to-$v_S$ paths for $S\in\cC$ has $\spr{\cC}$ edges. A closed DFS walk traverses each edge twice, so its length is $2\cdot\frac12\cdot\spr{\cC}=\spr{\cC}$ and it covers groups of total weight $\spr{\cov\br{\cC}}$. Thus the reduction preserves the optimum coverage for each budget. Since $\spr{V}=m+1$, Theorem~\ref{thm:gso-tree-source} gives the $\ProblemBPCSC$ guarantee.

    For $\ProblemPCSC$, run the same construction for every integer budget in $[m]$ and return a smallest family covering at least $f\cdot n/4$ elements among the outputs and the empty family. Let $\cCopt$ be an optimal $\ProblemPCSC$ solution. For $\spr{\cCopt}>0$, the budget $\spr{\cCopt}$ yields such a family of size $\bigo\br{\log m}\cdot\spr{\cCopt}$. The $\ProblemPCMSSC$ guarantee follows from Theorem~\ref{thm:BPCSC-to-PCMSSC}.
\end{proof}
